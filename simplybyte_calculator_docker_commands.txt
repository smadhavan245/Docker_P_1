2. RUN THE CONTAINER AND TRY VOLUME MOUNTS
------------------------------------------
# First attempt: failed because the container name already existed
docker run -d -p 9000:5000 -v hm:/volume --name calci-calcu calci-new

# The history shows several subsequent attempts with the same container name.
# These exact commands were entered:
docker run -d -p 9000:5000 -v hm:/volume --name calci-calcu calci-new
docker run -d -p 9000:5000 -v hm:/volume --name calci-calcu calci-new
docker run -d -p 9000:5000 -vC:\Users\Madhavan.S\Downloads\hm/volume --name calci-calcu calci-new

# Note:
# Docker reported a container-name conflict for calci-calcu.
# The -v syntax in the two Windows-path attempts is malformed/ambiguous.
# Correct named-volume syntax:
docker volume create hm
docker run -d -p 9000:5000 --mount source=hm,target=/volume --name calci-calcu calci-new

# If calci-calcu already exists, inspect it first:
docker ps -a
docker inspect calci-calcu

# If you intentionally want to replace that container (this removes the container,
# but does not remove the image or named volume):

docker run -d -p 9000:5000 --mount source=hm,target=/volume --name calci-calcu calci-new
